# Historical source only. Use the current operator manifest for execution.
param([Parameter(Mandatory=$true)][string]$OutDir)
$ErrorActionPreference='Stop'
New-Item -ItemType Directory -Force $OutDir | Out-Null
Add-Type -AssemblyName UIAutomationClient
Add-Type -AssemblyName System.Windows.Forms
. (Join-Path $PSScriptRoot '..\guards\kv_ui_guard.ps1')
$env:KV_WORKFLOW_RUN_LOG='H:\kvOp\error_snapshot_fix_20260906\sample_snapshot\run.log'
Initialize-KvUiGuard -OutDir $OutDir
$p=Get-Process Kvs
$root=[Windows.Automation.AutomationElement]::FromHandle($p.MainWindowHandle)
$pane=@($root.FindAll([Windows.Automation.TreeScope]::Descendants,[Windows.Automation.Condition]::TrueCondition)|Where-Object {$_.Current.AutomationId -eq '_tabFBMacroParam' -and $_.Current.ControlType -eq [Windows.Automation.ControlType]::Pane})
if($pane.Count -ne 1){throw 'Argument pane not unique'}
$filter=$pane[0].FindFirst([Windows.Automation.TreeScope]::Descendants,(New-Object Windows.Automation.PropertyCondition([Windows.Automation.AutomationElement]::AutomationIdProperty,'_usageFilterComboBox')))
Assert-KvUiForegroundHwnd -ExpectedHwnd $p.MainWindowHandle -Step 'foreground before argument filter focus' -ExpectedTitleLike 'KV STUDIO*' -AllowSingleRecovery|Out-Null
$filter.SetFocus()
Invoke-KvGuardedSendKeys -TargetHwnd $p.MainWindowHandle -Step 'all argument usage filter' -Keys '{HOME}{ENTER}' -ExpectedTitleLike 'KV STUDIO*'
Invoke-KvGuardedSendKeys -TargetHwnd $p.MainWindowHandle -Step 'focus argument filter accelerator' -Keys '%l' -ExpectedTitleLike 'KV STUDIO*'
Invoke-KvGuardedSendKeys -TargetHwnd $p.MainWindowHandle -Step 'argument filter to native cell' -Keys '+{TAB}' -ExpectedTitleLike 'KV STUDIO*'
$focus=[Windows.Automation.AutomationElement]::FocusedElement
[pscustomobject]@{id=$focus.Current.AutomationId;class=$focus.Current.ClassName;hwnd=$focus.Current.NativeWindowHandle;type=$focus.Current.ControlType.ProgrammaticName}|ConvertTo-Json|Set-Content (Join-Path $OutDir 'focus.json') -Encoding UTF8
$walker=[Windows.Automation.TreeWalker]::ControlViewWalker
$e=$focus;$owned=$false
while($e){if($e.Equals($pane[0])){$owned=$true;break};$e=$walker.GetParent($e)}
if(-not $owned){throw 'Argument pane does not own focused cell'}
Invoke-KvGuardedClipboardSetText -TargetHwnd $p.MainWindowHandle -Step 'argument snapshot sentinel' -Text '__EMPTY_COPY_SENTINEL__' -ExpectedTitleLike 'KV STUDIO*'
Invoke-KvGuardedCtrlChord -TargetHwnd $p.MainWindowHandle -Step 'select all argument cells' -Vk 0x41 -ExpectedTitleLike 'KV STUDIO*'
Invoke-KvGuardedCtrlChord -TargetHwnd $p.MainWindowHandle -Step 'copy all argument cells' -Vk 0x43 -ExpectedTitleLike 'KV STUDIO*'
[Windows.Forms.Clipboard]::GetText()|Set-Content (Join-Path $OutDir 'arguments.tsv') -Encoding UTF8
