;-- Definition  : @MR0 // when it is ON, sending of text data will be performed. 
;-------------------------------------------------------------------------------
;-- Id          : v 1.0 2012/12/17
;-------------------------------------------------------------------------------
;<h1/>Example for sending/receiving text data(Ladder) 
;To perform text data sending and receiving via Port0 
;Initialization will be performed and communication enabled when operation is started. 
;When using Port1, please change the related devices of Port0 to those of Port1. 
LD @CR2008
SET CR6105
CON
SET CR6100
;Send text data "STX KEYENCE ETX"
LDP @MR000
MOV #9 CM5101
MOV $024B CM5102
SMOV "EYENCE" CM5103
MOV $0300 CM5106
SET CR6104
;Receive text data (for monitoring confirmation) 
LD CR6101
BMOV CM5002 DM1000 CM5001
RES CR6101
;R500 is ON when a communication error occurs 
LD CR6002
OUT R500
;When a receive processing error occurs, output R501 will be ON.
LD CR6001
OUT R501
END
